import pandas as pd
from pathlib import Path
from typing import Optional, Union

class DataLoader:
    """支持多种格式的数据加载器"""

    @staticmethod
    def load(file_path: str, **kwargs) -> pd.DataFrame:
        """根据文件扩展名自动选择读取方法"""
        path = Path(file_path)
        suffix = path.suffix.lower()

        if suffix == '.csv':
            return pd.read_csv(file_path, **kwargs)
        elif suffix in ('.xls', '.xlsx'):
            return pd.read_excel(file_path, **kwargs)
        elif suffix == '.json':
            return pd.read_json(file_path, **kwargs)
        elif suffix == '.parquet':
            return pd.read_parquet(file_path, **kwargs)
        else:
            raise ValueError(f"不支持的文件格式: {suffix}")

    @staticmethod
    def load_large_csv(file_path: str, chunksize: int = 10000):
        """分块读取大文件，返回迭代器"""
        return pd.read_csv(file_path, chunksize=chunksize)

import pandas as pd
import numpy as np
from typing import Dict, List, Optional

class DataCleaner:
    """数据清洗引擎"""

    def __init__(self, config: Optional[Dict] = None):
        self.config = config or {}
        self.missing_threshold = self.config.get('missing_threshold', 0.3)  # 30%以上缺失则删除列

    def clean(self, df: pd.DataFrame) -> pd.DataFrame:
        """执行完整清洗流程"""
        df = self._remove_duplicates(df)
        df = self._handle_missing(df)
        df = self._standardize_dtypes(df)
        df = self._handle_outliers(df)
        return df

    def _remove_duplicates(self, df: pd.DataFrame) -> pd.DataFrame:
        return df.drop_duplicates()

    def _handle_missing(self, df: pd.DataFrame) -> pd.DataFrame:
        # 删除缺失率过高的列
        missing_ratio = df.isnull().mean()
        cols_to_drop = missing_ratio[missing_ratio > self.missing_threshold].index.tolist()
        df = df.drop(columns=cols_to_drop)

        # 数值列用中位数填充，类别列用众数填充
        for col in df.columns:
            if df[col].dtype in ['float64', 'int64']:
                df[col].fillna(df[col].median(), inplace=True)
            else:
                df[col].fillna(df[col].mode()[0] if not df[col].mode().empty else 'Unknown', inplace=True)
        return df

    def _standardize_dtypes(self, df: pd.DataFrame) -> pd.DataFrame:
        # 尝试将字符串列转为日期
        for col in df.select_dtypes(include=['object']).columns:
            try:
                df[col] = pd.to_datetime(df[col])
            except:
                pass
        return df

    def _handle_outliers(self, df: pd.DataFrame) -> pd.DataFrame:
        # 使用 IQR 方法处理异常值（超出 1.5 倍四分位距）
        for col in df.select_dtypes(include=['float64', 'int64']).columns:
            Q1 = df[col].quantile(0.25)
            Q3 = df[col].quantile(0.75)
            IQR = Q3 - Q1
            lower = Q1 - 1.5 * IQR
            upper = Q3 + 1.5 * IQR
            df[col] = df[col].clip(lower, upper)
        return df

    def get_cleaning_report(self, original: pd.DataFrame, cleaned: pd.DataFrame) -> Dict:
        """生成清洗报告"""
        return {
            'rows_removed': len(original) - len(cleaned),
            'cols_removed': len(original.columns) - len(cleaned.columns),
            'missing_handled': original.isnull().sum().sum() - cleaned.isnull().sum().sum(),
        }

import pandas as pd
import numpy as np
from typing import Dict, List

class StatAnalyzer:
    """统计分析引擎"""

    def describe(self, df: pd.DataFrame) -> Dict:
        """生成描述性统计"""
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        if len(numeric_cols) == 0:
            return {}

        desc = df[numeric_cols].describe().to_dict()
        # 添加偏度、峰度
        skewness = df[numeric_cols].skew().to_dict()
        kurtosis = df[numeric_cols].kurtosis().to_dict()
        return {
            'basic_stats': desc,
            'skewness': skewness,
            'kurtosis': kurtosis,
        }

    def correlation(self, df: pd.DataFrame, method: str = 'pearson') -> pd.DataFrame:
        """计算相关性矩阵"""
        numeric_df = df.select_dtypes(include=[np.number])
        if numeric_df.shape[1] < 2:
            return pd.DataFrame()
        return numeric_df.corr(method=method)

    def detect_outliers_iqr(self, df: pd.DataFrame, col: str) -> Dict:
        """检测某列的异常值"""
        Q1 = df[col].quantile(0.25)
        Q3 = df[col].quantile(0.75)
        IQR = Q3 - Q1
        lower = Q1 - 1.5 * IQR
        upper = Q3 + 1.5 * IQR
        outliers = df[(df[col] < lower) | (df[col] > upper)]
        return {
            'count': len(outliers),
            'lower_bound': lower,
            'upper_bound': upper,
            'ratio': len(outliers) / len(df)
        }

import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd
from pathlib import Path
from typing import Optional, List

class VizGenerator:
    """自动生成图表"""

    def __init__(self, output_dir: str = "workspace/agent/reports/data-expert"):
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)
        # 设置中文字体（避免乱码）
        plt.rcParams['font.sans-serif'] = ['SimHei', 'DejaVu Sans']
        plt.rcParams['axes.unicode_minus'] = False

    def auto_generate(self, df: pd.DataFrame, prefix: str = "analysis") -> List[str]:
        """根据数据类型自动生成图表"""
        saved_files = []
        numeric_cols = df.select_dtypes(include=['float64', 'int64']).columns
        categorical_cols = df.select_dtypes(include=['object', 'category']).columns

        # 1. 数值列分布直方图（最多5个）
        for col in numeric_cols[:5]:
            path = self._plot_histogram(df, col, prefix)
            if path:
                saved_files.append(path)

        # 2. 相关性热力图（如果有2个以上数值列）
        if len(numeric_cols) >= 2:
            path = self._plot_correlation_heatmap(df, numeric_cols, prefix)
            if path:
                saved_files.append(path)

        # 3. 分类列计数图（最多3个）
        for col in categorical_cols[:3]:
            if df[col].nunique() <= 20:  # 避免类别过多
                path = self._plot_countplot(df, col, prefix)
                if path:
                    saved_files.append(path)

        return saved_files

    def _plot_histogram(self, df: pd.DataFrame, col: str, prefix: str) -> Optional[str]:
        fig, ax = plt.subplots(figsize=(8, 5))
        df[col].hist(bins=30, ax=ax, edgecolor='black')
        ax.set_title(f'{col} 分布直方图')
        ax.set_xlabel(col)
        ax.set_ylabel('频数')
        filepath = self.output_dir / f"{prefix}_hist_{col}.png"
        plt.savefig(filepath, dpi=100, bbox_inches='tight')
        plt.close()
        return str(filepath)

    def _plot_correlation_heatmap(self, df: pd.DataFrame, cols: List[str], prefix: str) -> Optional[str]:
        corr = df[cols].corr()
        fig, ax = plt.subplots(figsize=(10, 8))
        sns.heatmap(corr, annot=True, fmt='.2f', cmap='coolwarm', ax=ax)
        ax.set_title('相关性热力图')
        filepath = self.output_dir / f"{prefix}_corr.png"
        plt.savefig(filepath, dpi=100, bbox_inches='tight')
        plt.close()
        return str(filepath)

    def _plot_countplot(self, df: pd.DataFrame, col: str, prefix: str) -> Optional[str]:
        fig, ax = plt.subplots(figsize=(10, 6))
        df[col].value_counts().head(15).plot(kind='bar', ax=ax)
        ax.set_title(f'{col} 类别计数')
        ax.set_xlabel(col)
        ax.set_ylabel('数量')
        plt.xticks(rotation=45, ha='right')
        filepath = self.output_dir / f"{prefix}_count_{col}.png"
        plt.savefig(filepath, dpi=100, bbox_inches='tight')
        plt.close()
        return str(filepath)
3.5 数据质量评分器（analyzers/score_calculator.py）
python
import pandas as pd
import numpy as np
from dataclasses import dataclass

@dataclass
class DataQualityScore:
    overall: float
    completeness: float   # 完整性
    consistency: float    # 一致性（格式/类型）
    accuracy: float       # 准确性（异常值）
    grade: str
    passed: bool

class ScoreCalculator:
    """数据质量评分"""

    WEIGHTS = {
        'completeness': 0.4,
        'consistency': 0.3,
        'accuracy': 0.3,
    }

    def calculate(self, df: pd.DataFrame) -> DataQualityScore:
        completeness = self._completeness_score(df)
        consistency = self._consistency_score(df)
        accuracy = self._accuracy_score(df)

        overall = (completeness * self.WEIGHTS['completeness'] +
                   consistency * self.WEIGHTS['consistency'] +
                   accuracy * self.WEIGHTS['accuracy'])

        grade = self._score_to_grade(overall)
        return DataQualityScore(
            overall=overall,
            completeness=completeness,
            consistency=consistency,
            accuracy=accuracy,
            grade=grade,
            passed=overall >= 60
        )

    def _completeness_score(self, df: pd.DataFrame) -> float:
        """完整性：非空值比例"""
        total_cells = df.shape[0] * df.shape[1]
        non_null_cells = df.notnull().sum().sum()
        return (non_null_cells / total_cells) * 100

    def _consistency_score(self, df: pd.DataFrame) -> float:
        """一致性：数据类型是否规范（数值列无字符串混入）"""
        score = 100.0
        for col in df.columns:
            if df[col].dtype == 'object':
                # 检查是否有明显的数字字符串
                numeric_ratio = df[col].str.match(r'^\d+(\.\d+)?$').mean() if len(df) > 0 else 0
                if numeric_ratio > 0.5:
                    score -= 10
        return max(0, score)

    def _accuracy_score(self, df: pd.DataFrame) -> float:
        """准确性：异常值比例（使用 Z-score）"""
        numeric_cols = df.select_dtypes(include=[np.number]).columns
        if len(numeric_cols) == 0:
            return 100.0

        outlier_ratios = []
        for col in numeric_cols:
            z_scores = np.abs((df[col] - df[col].mean()) / df[col].std())
            outlier_ratio = (z_scores > 3).mean()
            outlier_ratios.append(outlier_ratio)

        avg_outlier_ratio = np.mean(outlier_ratios) if outlier_ratios else 0
        return max(0, 100 * (1 - avg_outlier_ratio))

    def _score_to_grade(self, score: float) -> str:
        if score >= 90: return 'A'
        elif score >= 80: return 'B'
        elif score >= 70: return 'C'
        elif score >= 60: return 'D'
        else: return 'F'

            lines.append("2. 已自动删除重复行，请检查是否影响业务")
        if len(chart_paths) == 0:
            lines.append("3. 未能生成图表，请检查是否有数值列")

        lines.append("")
        lines.append("---")
        lines.append(f"*报告生成时间: {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}*")

        return "\n".join(lines)

    def save_report(self, content: str, file_name: str) -> str:
        """保存报告到文件"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        base = Path(file_name).stem
        report_path = self.output_dir / f"{base}_report_{timestamp}.md"
        report_path.write_text(content, encoding='utf-8')
        return str(report_path)

python
#!/usr/bin/env python
# -*- coding: utf-8 -*-
"""数据专家 Agent 入口脚本"""

import argparse
import sys
from pathlib import Path

# 添加当前目录到路径
sys.path.insert(0, str(Path(__file__).parent))

from collectors.data_loader import DataLoader
from analyzers.data_cleaner import DataCleaner
from analyzers.stat_analyzer import StatAnalyzer
from analyzers.viz_generator import VizGenerator
from analyzers.score_calculator import ScoreCalculator
from generators.report_generator import ReportGenerator

def run_analysis(file_path: str, visualize: bool = True):
    """执行完整数据分析流程"""
    print(f"📂 正在加载数据: {file_path}")
    df = DataLoader.load(file_path)
    print(f"✅ 加载成功: {df.shape[0]} 行, {df.shape[1]} 列")

    # 清洗
    print("🧹 开始数据清洗...")
    cleaner = DataCleaner()
    df_cleaned = cleaner.clean(df)
    cleaning_report = cleaner.get_cleaning_report(df, df_cleaned)
    print(f"   删除重复行: {cleaning_report['rows_removed']}, 删除列: {cleaning_report['cols_removed']}")

    # 统计分析
    print("📈 执行统计分析...")
    stat_analyzer = StatAnalyzer()
    stats = stat_analyzer.describe(df_cleaned)
    print(f"   数值列数: {len(stats.get('basic_stats', {}))}")

    # 数据质量评分
    print("🔍 计算数据质量评分...")
    scorer = ScoreCalculator()
    quality_score = scorer.calculate(df_cleaned)
    print(f"   综合评分: {quality_score.overall:.1f} ({quality_score.grade})")

    # 可视化
    chart_paths = []
    if visualize:
        print("📊 生成可视化图表...")
        viz = VizGenerator()
        chart_paths = viz.auto_generate(df_cleaned, prefix=Path(file_path).stem)
        print(f"   已生成 {len(chart_paths)} 个图表")

    # 生成报告
    print("📝 生成分析报告...")
    reporter = ReportGenerator()
    report_content = reporter.generate(
        file_name=Path(file_path).name,
        df_shape=df_cleaned.shape,
        stats=stats,
        quality_score=quality_score,
        cleaning_report=cleaning_report,
        chart_paths=chart_paths
    )
    report_path = reporter.save_report(report_content, file_path)
    print(f"✅ 报告已保存: {report_path}")

    return {
        'report_path': report_path,
        'charts': chart_paths,
        'quality_score': quality_score.overall,
        'quality_grade': quality_score.grade
    }

if __name__ == '__main__':
    parser = argparse.ArgumentParser(description='数据专家 Agent')
    parser.add_argument('--file', required=True, help='数据文件路径')
    parser.add_argument('--no-viz', action='store_true', help='不生成可视化')
    args = parser.parse_args()

    run_analysis(args.file, visualize=not args.no_viz)

六、技术要点总结
6.1 与代码审查助手的异同
维度	代码审查助手	数据专家 Agent
采集层	Git 克隆、本地扫描	文件加载（CSV/Excel等）
分析层	Ruff/Radon/Bandit	清洗/统计/可视化/评分
报告层	Markdown + 飞书	Markdown + 图表文件
评分体系	质量/安全/复杂度/风格	完整性/一致性/准确性
6.2 关键决策
使用 pandas 作为核心数据处理引擎，稳定且生态丰富。

自动图表选择：根据数据类型自动选择直方图、热力图、计数图，避免 Agent 写代码。

分块读取：大文件支持迭代器，防止内存溢出。

质量评分：量化数据可用性，给用户明确改进方向。

6.3 扩展方向
智能推荐分析：根据数据特征自动建议统计检验（t检验、ANOVA等）。

机器学习集成：自动训练简单预测模型（线性回归、聚类）。

增量分析：只分析新增数据，生成对比报告。

自然语言查询：用户问“哪个地区的销售额最高”，Agent 自动转化为 pandas 操作。

七、写在最后
借鉴代码审查助手的成功经验，我们构建的数据专家 Agent 具有以下特点：

分层清晰：采集 → 分析 → 报告，易于扩展新分析器。

开箱即用：支持常见文件格式，自动清洗和可视化。

质量可量化：数据质量评分让用户一眼知道数据是否可靠。

与 JiuwenSwarm 无缝集成：可作为 Skill 被 Agent 调用，也可作为命令行工具独立运行。

你可以将 data-expert 文件夹放置在 workspace/agent/skills/ 下，并在 SKILL.md 中配置好 allowed_tools，即可通过 JiuwenSwarm 的 Agent 调用来完成端到端的数据分析任务。
